#include <TimerOne.h>
#include <SPI.h>
#include <Ethernet.h>

//아두이노 IP, MAC
byte mac[]={0xAC, 0xAD, 0xBE, 0xEF, 0xFE, 0xED};
IPAddress ip(192, 168, 3, 21);

// PLC IP
IPAddress plc_ip(192, 168, 3, 39);
EthernetClient client;

const char* FRAME_3E = "5000";
const char* HOSTSTATION = "00FF03FF00";
const char* DATALENGTH = "0020";
const char* TIMER = "0010";

const char* CMD = "1401";
const char* SUBCMD="0000";
const char* DEVICE ="M*";
const char* DEVICENUM="000100";
const char* DEVICEPOINT="0002";

String cmd=String(FRAME_3E)+HOSTSTATION+DATALENGTH+TIMER+ CMD+SUBCMD+DEVICE+DEVICENUM+DEVICEPOINT;


const char* rear_data = "0000";//문자열

const int  lamp1_on = 0x0001;//숫자
const int  lamp1_off = ~(0x0001);//숫자=0xfffe

const int lamp2_on = 0x0002;
const int lamp2_off = ~(0x0002); //

const int lamp3_on = 0x0004;
const int lamp3_off = ~(0x0004); //

const int lamp4_on = 0x0008;
const int lamp4_off = ~(0x0008); //



union SWData { // 사용자 정의 데이터 타입
  uint8_t value;

  struct {
    uint8_t sw0 : 1;
    uint8_t sw1 : 1;
    uint8_t sw2 : 1;
    uint8_t sw3 : 1;
  } bit;
};

SWData tmpSW;

void setup() {
 Ethernet.begin(mac,ip); 
 Serial.begin(9600);
 Timer1.initialize(500000);
 Timer1.attachInterrupt(Timer_fnc);

 pinMode(30,OUTPUT);
 pinMode(32,OUTPUT);
 pinMode(34,OUTPUT);
 pinMode(36,OUTPUT);

 pinMode(22,INPUT_PULLUP);
 pinMode(24,INPUT_PULLUP);
 pinMode(26,INPUT_PULLUP);
 pinMode(28,INPUT_PULLUP);
 

 digitalWrite(30,HIGH);
 digitalWrite(32,HIGH);
 digitalWrite(34,HIGH);
 digitalWrite(36,HIGH);

 if(client.connect(plc_ip, 6001)){
    Serial.println("Connected to PLC");

 }
 else{
    Serial.println("Connection failed");
 }

}
int i;
int sw1, sw2, sw3, sw4;
int state;
void Timer_fnc()
{
  tmpSW.value = 0; //초기화 함

  sw1 = digitalRead(22);
  sw2 = digitalRead(24);
  sw3 = digitalRead(26);
  sw4 = digitalRead(28);
  
    if(sw1==0){
      tmpSW.bit.sw0 = 1;
    }
    if(sw2==0){
      tmpSW.bit.sw1 = 1;
    }
     
    if(sw3==0){
      tmpSW.bit.sw2 = 1;
    }
    if(sw4==0){
      tmpSW.bit.sw3 = 1;
    }
  
  switch(tmpSW.value){
    case 1:
      state = 100;
    break;
    case 2:
      state = 200;
    break;
    case 4:
      state = 300;
    break;
    case 8:
      state = 400;
    break;

  }
  
}

int final_data;

int sw1_cnt, sw2_cnt, sw3_cnt, sw4_cnt;

char final_str[5];//배열로 선언

void loop() {
 
 String msg;
 
 switch(state){
  case 100: //lamp1_on
    sw1_cnt++;

    if(sw1_cnt%2){
      digitalWrite(30,LOW);
      final_data |=  lamp1_on; //특정 M100비튼 만 ON
      sprintf(final_str,"%04X",final_data);
      msg = cmd+final_str+ rear_data;
      client.print(msg);
    }
    else{
      digitalWrite(30,HIGH);
      final_data &=  lamp1_off; //특정 M100비튼 만 ON
      sprintf(final_str,"%04X",final_data);
      msg = cmd+final_str+ rear_data;
      client.print(msg);
    }
    
    state = 500;
  break;

  case 200: //lamp1_off
    sw2_cnt++;

    if(sw2_cnt%2){
      digitalWrite(32,LOW);
      final_data |=  lamp2_on; //특정 M100비튼 만 ON
      sprintf(final_str,"%04X",final_data);
      msg = cmd+final_str+ rear_data;
      client.print(msg);
    }
    else{
      digitalWrite(32,HIGH);
      final_data &=  lamp2_off; //특정 M100비튼 만 ON
      sprintf(final_str,"%04X",final_data);
      msg = cmd+final_str+ rear_data;
      client.print(msg);
    }


    state = 500;
  break;

  case 300:
    sw3_cnt++;

    if(sw3_cnt%2){
      digitalWrite(34,LOW);
      final_data |=  lamp3_on; //특정 M100비튼 만 ON
      sprintf(final_str,"%04X",final_data);
      msg = cmd+final_str+ rear_data;
      client.print(msg);
    }
    else{
      digitalWrite(34,HIGH);
      final_data &=  lamp3_off; //특정 M100비튼 만 ON
      sprintf(final_str,"%04X",final_data);
      msg = cmd+final_str+ rear_data;
      client.print(msg);
    }
    state = 500;
  break;

  case 400:
    sw4_cnt++;

    if(sw4_cnt%2){
      digitalWrite(36,LOW);
      final_data |=  lamp4_on; //특정 M100비튼 만 ON
      sprintf(final_str,"%04X",final_data);
      msg = cmd+final_str+ rear_data;
      client.print(msg);
    }
    else{
      digitalWrite(36,HIGH);
      final_data &=  lamp4_off; //특정 M100비튼 만 ON
      sprintf(final_str,"%04X",final_data);
      msg = cmd+final_str+ rear_data;
      client.print(msg);
    }
    state = 500;
  break;

  case 500:
  break;

 }

 if(client.available()){
  char c = client.read();
  Serial.print(c);
 }

}
