#include <TimerOne.h>

#include <SPI.h>
#include <Ethernet.h>

// 아두이노 IP, MAC 주소
byte mac[] = { 0xDE, 0xAD, 0xBE, 0xEF, 0xFE, 0xED };
IPAddress ip(192, 168, 3, 20); // 아두이노 IP
IPAddress plc_ip(192, 168, 3, 39); // PLC IP
EthernetClient client;

// 아두이노가 쓸 때
const char* FRAME_3E = "5000";
const char* HOSTSTATION = "00FF03FF00";
const char* DATALENGTH_R = "0018"; // For Write , 참고 "0018" : For Read
const char* DATALENGTH_W = "0020"; // For Write , 참고 "0018" : For Read

const char* TIMER = "0010";


const char* CMD_R ="0401";//For Write, 참고 "0401" : For Read
const char* CMD_W ="1401";//For Write, 참고 "0401" : For Read

const char* SUBCMD = "0000";
const char* DEVICE_R = "X*";
const char* DEVICE_W = "Y*";

const char* DEVICENUM="000100";//
const char* DEVICEPOINT = "0002";

String cmd_read = String(FRAME_3E)+ HOSTSTATION+DATALENGTH_R+TIMER+CMD_R+SUBCMD+DEVICE_R+DEVICENUM+ DEVICEPOINT;
String cmd_write = String(FRAME_3E)+ HOSTSTATION+DATALENGTH_W+TIMER+CMD_W+SUBCMD+DEVICE_W+DEVICENUM+ DEVICEPOINT;



void setup() {
  Serial.begin(9600);

  Ethernet.begin(mac, ip); 
  delay(1000);

 
 
  pinMode(22,INPUT_PULLUP);
  pinMode(24,INPUT_PULLUP);
  pinMode(26,INPUT_PULLUP);
  pinMode(28,INPUT_PULLUP);
  
  pinMode(30,OUTPUT);
  pinMode(32,OUTPUT);
  pinMode(34,OUTPUT);
  pinMode(36,OUTPUT);
  //초기화(LED all off)
  digitalWrite(30,HIGH);
  digitalWrite(32,HIGH);
  digitalWrite(34,HIGH);
  digitalWrite(36,HIGH);

  if (client.connect(plc_ip, 6001)) {
    Serial.println("Connected to PLC");           
    }

  else {
    Serial.println("Connection failed");
    client.stop();
    delay(1000);
  }
  
}

int state=100;
int num=0;

void loop() {
  String msg;
  char str_write[5];
  String response="";
  
    switch(state){
     
      case 100:{// 지역 변수를 구분하기 위해 {}를 사용함! Read Command Sending
        //digitalWrite(30,LOW);    
        if(client.connected()){ // 연결되었는가? 연결되었다면 다음 상태는 state=200
          msg = cmd_read;
          client.print(msg);            
          state=200;// 연속적으로 명령어가 나가지 않도록 처리함
        }
        else{//연결되지 않았다면 또 기다린다는 의미 state=600
          state = 600;
        }
        break;
      }
      case 200:{
           while(client.available()) { // 수신 
            char c = client.read();
            response +=c;    
        }
        //Serial.println("RECEIVE COMPLETE");
        
        if(response.length() >= 26) {
            String SubStr = response.substring(22,26);
            num =(int) strtol(SubStr.c_str(),NULL,16);
            //Serial.print(SubStr); Serial.print(" =*=> "); Serial.println(num);
        }
        state=300;
      }
      case 300:{           
         if(num&0x01){   digitalWrite(30,LOW); }
         if(!(num&0x01)){ digitalWrite(30,HIGH);}
         if(num&0x02)  { digitalWrite(32,LOW);}
         if(!(num&0x02)){ digitalWrite(32,HIGH);}
         if(num&0x04)   {digitalWrite(34,LOW);}
         if(!(num&0x04)) {digitalWrite(34,HIGH);}
         if(num&0x08)   {digitalWrite(36,LOW);}
         if(!(num&0x08)) {digitalWrite(36,HIGH);}
         
         state = 400;
        break;
      }

      case 400:{
         sprintf(str_write,"%04X",num);
         msg = cmd_write+str_write+"0000";
        client.print(msg);            
        state=500;
        break;
      }
      case 500:{
        delay(10);
        state=100;
        break;
      }
      case 600:{
        //delay(2);
        state=100;
        break;
      }
    }
    // Background

    
}
