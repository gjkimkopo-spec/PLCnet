#include <TimerOne.h>

void setup() {

 Timer1.initialize(10000);
 Timer1.attachInterrupt(Timer_fnc);

 pinMode(30,OUTPUT);
 pinMode(32,OUTPUT);
 pinMode(34,OUTPUT);
 pinMode(36,OUTPUT);

 pinMode(22,INPUT_PULLUP);
 pinMode(24,INPUT_PULLUP);
 pinMode(26,INPUT_PULLUP);
 pinMode(28,INPUT_PULLUP);
 

 digitalWrite(30,HIGH);
 digitalWrite(32,HIGH);
 digitalWrite(34,HIGH);
 digitalWrite(36,HIGH);

}
int i;
int sw1, sw2, sw3, sw4;
int state;
void Timer_fnc()
{
  sw1 = digitalRead(22);
  sw2 = digitalRead(24);
 
  
  if(sw1==1 && sw2==1 && sw3==1 && sw4==1 ){
    // 인식할 필요가 없음(스위치가 눌리지 않았기 때문)
    state = 0;
  }
  else{
    if(sw1==0){
      state = 100;
    }
    if(sw2==0){
      state = 200;
    }  
  }
}

void loop() {
 
 switch(state){
  case 100:
    digitalWrite(30,LOW);
    digitalWrite(32,HIGH);
    state = 500;
  break;

  case 200:
    digitalWrite(30,HIGH);
    digitalWrite(32,LOW);
    state = 500;
  break;

  case 500:
  break;

 }

}
